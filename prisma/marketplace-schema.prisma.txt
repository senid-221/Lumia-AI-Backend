Marketplace domain design (merge these models into prisma/schema.prisma during the next migration):

enum MerchantStatus { PENDING ACTIVE SUSPENDED }
enum ProductSourceType { MANUAL API CSV PARTNER }
enum ChannelType { WHATSAPP INSTAGRAM FACEBOOK TIKTOK }
enum ChannelStatus { DISCONNECTED CONNECTED ERROR }
enum PayoutStatus { PENDING PROCESSING PAID FAILED }

model Merchant {
  id            String          @id @default(cuid())
  businessId    String
  displayName   String
  status        MerchantStatus  @default(PENDING)
  contactPhone  String?
  contactEmail  String?
  createdAt     DateTime        @default(now())
  updatedAt     DateTime        @updatedAt
  business      Business        @relation(fields:[businessId], references:[id], onDelete:Cascade)
  stores        MerchantStore[]
  sources       ProductSource[]
  products      MarketplaceProduct[]
  commissions   Commission[]
  payouts       Payout[]
  @@index([businessId,status])
}

model MerchantStore {
  id          String   @id @default(cuid())
  merchantId  String
  name        String
  platform    String?
  externalId  String?
  storeUrl    String?
  active      Boolean  @default(true)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  merchant    Merchant @relation(fields:[merchantId], references:[id], onDelete:Cascade)
  sources     ProductSource[]
  @@index([merchantId,active])
}

model ProductSource {
  id            String           @id @default(cuid())
  merchantId    String
  storeId       String?
  type          ProductSourceType
  name          String
  endpoint      String?
  active        Boolean          @default(true)
  lastSyncedAt  DateTime?
  createdAt     DateTime         @default(now())
  updatedAt     DateTime         @updatedAt
  merchant      Merchant         @relation(fields:[merchantId], references:[id], onDelete:Cascade)
  store         MerchantStore?   @relation(fields:[storeId], references:[id], onDelete:SetNull)
  products      MarketplaceProduct[]
  @@index([merchantId,active])
}

model MarketplaceProduct {
  id              String          @id @default(cuid())
  merchantId      String
  sourceId        String?
  externalId      String?
  sku             String?
  name            String
  description     String?
  imageUrl        String?
  costPrice       Decimal         @db.Decimal(14,2)
  sellingPrice    Decimal         @db.Decimal(14,2)
  currency        String          @default("RWF")
  stock           Int             @default(0)
  commissionRate  Decimal         @db.Decimal(7,4) @default(0)
  active          Boolean         @default(true)
  lastSyncedAt    DateTime?
  createdAt       DateTime        @default(now())
  updatedAt       DateTime        @updatedAt
  merchant        Merchant       @relation(fields:[merchantId], references:[id], onDelete:Cascade)
  source          ProductSource? @relation(fields:[sourceId], references:[id], onDelete:SetNull)
  commissions     Commission[]
  @@index([merchantId,active])
  @@index([name])
}

model Commission {
  id            String            @id @default(cuid())
  merchantId    String
  productId     String
  orderId       String
  grossAmount   Decimal           @db.Decimal(14,2)
  commission   Decimal           @db.Decimal(14,2)
  merchantNet   Decimal           @db.Decimal(14,2)
  currency      String            @default("RWF")
  createdAt     DateTime          @default(now())
  merchant      Merchant          @relation(fields:[merchantId], references:[id], onDelete:Cascade)
  product       MarketplaceProduct @relation(fields:[productId], references:[id])
  @@index([merchantId,createdAt])
}

model Payout {
  id          String        @id @default(cuid())
  merchantId  String
  amount      Decimal       @db.Decimal(14,2)
  currency    String        @default("RWF")
  status      PayoutStatus  @default(PENDING)
  reference   String?
  paidAt      DateTime?
  createdAt   DateTime      @default(now())
  updatedAt   DateTime      @updatedAt
  merchant    Merchant      @relation(fields:[merchantId], references:[id], onDelete:Cascade)
  @@index([merchantId,status])
}

model SocialChannel {
  id             String        @id @default(cuid())
  businessId     String
  type           ChannelType
  status         ChannelStatus @default(DISCONNECTED)
  externalPageId String?
  externalUserId String?
  accessTokenRef String?
  metadata       Json?
  connectedAt    DateTime?
  createdAt      DateTime      @default(now())
  updatedAt      DateTime      @updatedAt
  business       Business      @relation(fields:[businessId], references:[id], onDelete:Cascade)
  @@unique([businessId,type])
  @@index([businessId,status])
}

model SocialLead {
  id             String       @id @default(cuid())
  businessId     String
  channelId      String
  customerId     String?
  externalUserId String?
  sourceUrl      String?
  campaign       String?
  createdAt      DateTime     @default(now())
  business       Business     @relation(fields:[businessId], references:[id], onDelete:Cascade)
  channel        SocialChannel @relation(fields:[channelId], references:[id], onDelete:Cascade)
  @@index([businessId,createdAt])
}

Migration note: Order/OrderItem should later gain optional marketplace metadata (merchantId, marketplaceProductId, source) or a separate MarketplaceOrder/MarketplaceOrderItem layer to support multi-vendor checkout without breaking the existing business order model.
